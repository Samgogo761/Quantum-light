```bash
python -B tools/b0_small_model/q8_m0_gamma12_runner_20261006.py prepare \
  --run-root "$RUN" \
  --fixture results/q_line/p1_h2_finite_kcv_gamma12_backend_cuda_resume_20261006/backend_fixture.npz

  --run-root "$RUN" \

  --backend cpu \
  --window start \
  --block-id cpu16 \
  --column-start 0 \
  --column-stop 256
```
